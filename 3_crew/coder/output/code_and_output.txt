```python
# Python code to calculate the first 10,000 terms of the series
# 1 - 1/3 + 1/5 - 1/7 + ... and multiply the total by 4

total = 0
for i in range(10000):
    term = 1 / (2 * i + 1)
    if i % 2 == 0:
        total += term
    else:
        total -= term

result = total * 4
print(result)
```
